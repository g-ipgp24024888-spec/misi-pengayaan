const questions = [
{
q:"Suasana di pasar malam sangat ______ apabila ramai pengunjung berbual dan peniaga memanggil pelanggan.",
a:"Riuh",
options:["Riuh","Senyap","Garau","Merdu"]
},
{
q:"Bunga ros yang diletakkan di atas meja mengeluarkan bau yang ______.",
a:"Harum",
options:["Hanyir","Harum","Basi","Busuk"]
},
{
q:"Permukaan lantai yang baru digilap kelihatan ______ apabila terkena cahaya.",
a:"Licin",
options:["Kasar","Kesat","Licin","Keras"]
},
{
q:"Suara penyanyi itu sangat ______ sehingga penonton terpukau mendengar nyanyiannya.",
a:"Merdu",
options:["Bising","Garau","Nyaring","Merdu"]
},
{
q:"Sup yang dimasak oleh ibu terasa sangat ______ kerana mempunyai rasa yang enak dan berlemak.",
a:"Lemak",
options:["Masin","Lemak","Pahit","Hanyir"]
},
{
q:"Bentuk ukiran pada bangunan bersejarah itu sangat ______ dan menarik perhatian pelancong.",
a:"Indah",
options:["Buruk","Comel","Indah","Garau"]
},
{
q:"Jeritan kanak-kanak itu kedengaran sangat ______ sehingga mengejutkan orang di sekeliling.",
a:"Nyaring",
options:["Senyap","Nyaring","Merdu","Harum"]
},
{
q:"Ikan yang tidak disimpan dengan baik telah mengeluarkan bau ______.",
a:"Busuk",
options:["Wangi","Harum","Busuk","Nyaman"]
},
{
q:"Batu yang digunakan untuk membina tembok itu sangat ______ dan sukar dialihkan.",
a:"Keras",
options:["Lembut","Halus","Licin","Keras"]
},
{
q:"Kuih yang disediakan oleh nenek mempunyai rasa yang ______ kerana gula melaka yang digunakan.",
a:"Manis",
options:["Pedas","Manis","Masin","Pahit"]
}
];

const matching = document.getElementById("matching");

questions.forEach((item, i) => {
    let options = [...item.options].sort(() => Math.random() - 0.5);

    matching.innerHTML += `
    <div class="question">
        <b>${i+1}. ${item.q}</b>
        <select id="answer${i}">
            <option value="">-- Pilih jawapan --</option>
            ${options.map(o => `<option value="${o}">${o}</option>`).join("")}
        </select>
    </div>`;
});

function checkMatching() {
    let score = 0;
    let unanswered = 0;

    questions.forEach((item, i) => {
        const answer = document.getElementById("answer"+i).value;
        if (!answer) unanswered++;
        if (answer === item.a) score++;
    });

    const result = document.getElementById("matchResult");
    result.style.display = "block";

    if (unanswered > 0) {
        result.className = "result wrong";
        result.innerHTML = `Sila jawab semua soalan dahulu. ${unanswered} soalan belum dijawab.`;
        return;
    }

    result.className = "result correct";
    result.innerHTML = `⭐ Markah anda: ${score}/10<br>` +
        (score === 10 ? "Hebat! Semua jawapan anda tepat!" :
        score >= 7 ? "Bagus! Teruskan usaha anda!" :
        "Jangan berputus asa. Cuba teliti semula maksud setiap kata adjektif.");
}

function resetMatching() {
    questions.forEach((item, i) => {
        document.getElementById("answer"+i).value = "";
    });
    document.getElementById("matchResult").style.display = "none";
}

const tasks = [
{
sense:"👁️ Deria Pandang",
word:"Indah",
prompt:"Pemandangan matahari terbenam di tepi pantai."
},
{
sense:"👂 Deria Dengar",
word:"Merdu",
prompt:"Suara penyanyi semasa persembahan kebudayaan."
},
{
sense:"👃 Deria Bau",
word:"Hanyir",
prompt:"Bau ikan yang baru dibawa pulang dari pasar."
},
{
sense:"👅 Deria Rasa",
word:"Lazat",
prompt:"Hidangan istimewa yang disediakan oleh ibu."
},
{
sense:"✋ Deria Sentuh",
word:"Lembut",
prompt:"Bulu arnab yang disentuh oleh seorang kanak-kanak."
}
];

const sentenceTasks = document.getElementById("sentenceTasks");

tasks.forEach((task, i) => {
    sentenceTasks.innerHTML += `
    <div class="question">
        <h3>${task.sense}</h3>
        <p><b>Kata adjektif:</b>
        <span class="word">${task.word}</span></p>
        <p><b>Situasi:</b> ${task.prompt}</p>
        <label><b>Ayat anda:</b></label>
        <textarea id="sentence${i}" placeholder="Tulis ayat lengkap anda di sini..."></textarea>
        <p style="font-size:13px;color:#64748b;">
        Bilangan perkataan: <span id="count${i}">0</span>
        </p>
    </div>`;
});

tasks.forEach((task, i) => {
    document.getElementById("sentence"+i).addEventListener("input", function() {
        let count = this.value.trim() ? this.value.trim().split(/\s+/).length : 0;
        document.getElementById("count"+i).textContent = count;
    });
});

function checkSentences() {
    let completed = 0;
    let feedback = [];

    tasks.forEach((task, i) => {
        let sentence = document.getElementById("sentence"+i).value.trim();
        let words = sentence ? sentence.split(/\s+/) : [];

        let hasWord = new RegExp("\\b" + task.word + "\\b", "i").test(sentence);
        let enoughWords = words.length >= 8;
        let hasCapital = /^[A-ZÀ-Ý]/.test(sentence);
        let hasPunctuation = /[.!?]$/.test(sentence);

        if (hasWord && enoughWords && hasCapital && hasPunctuation) {
            completed++;
            feedback.push(`✅ Cabaran ${i+1}: Lengkap.`);
        } else {
            let issues = [];
            if (!hasWord) issues.push("Gunakan kata adjektif yang diberikan.");
            if (!enoughWords) issues.push("Ayat perlu mempunyai sekurang-kurangnya 8 patah perkataan.");
            if (!hasCapital) issues.push("Mulakan ayat dengan huruf besar.");
            if (!hasPunctuation) issues.push("Akhiri ayat dengan tanda baca yang sesuai.");

            feedback.push(`❗ Cabaran ${i+1}: ${issues.join(" ")}`);
        }
    });

    const result = document.getElementById("sentenceResult");
    result.style.display = "block";
    result.className = "result " + (completed === 5 ? "correct" : "wrong");

    result.innerHTML = `
    <h3>Keputusan Cabaran</h3>
    <p>${completed}/5 ayat memenuhi syarat asas.</p>
    ${feedback.map(f => `<p>${f}</p>`).join("")}
    <p><b>Ingat:</b> Semak juga tatabahasa, kesesuaian maksud dan kejelasan ayat anda.</p>
    `;
}
